Routes

(A) - Admin Only
(P) - Public
(L) - Logged In
(V) - Admin Or In Your Account

--Static Pages--
                root        /                              (P)  static#home
                help        /help(.:format)                (P)  static#help
                home        /home(.:format)                (P)  static#home
               admin        /admin(.:format)               (A)  static#admin

 --Sessions--              
    new_user_session GET    /users/sign_in(.:format)       (P)  devise/sessions#new
        user_session POST   /users/sign_in(.:format)       (P)  devise/sessions#create
destroy_user_session DELETE /users/sign_out(.:format)      (L)  devise/sessions#destroy

--User Accounts--
               users GET    /users(.:format)               (A)  users#index
                     POST   /users(.:format)               (P)  users#create
            new_user GET    /users/new(.:format)           (P)  users#new
           edit_user GET    /users/:id/edit(.:format)      (V)  users#edit
                user GET    /users/:id(.:format)           (V)  users#show
                     PUT    /users/:id(.:format)           (V)  users#update
                     DELETE /users/:id(.:format)           (V)  users#destroy
--Groups--

           new_group GET    /groups/new(.:format)          (A)  groups#new
          edit_group GET    /groups/:id/edit(.:format)     (A)  groups#edit
               group GET    /groups/:id(.:format)          (A)  groups#show
